cmake_minimum_required(VERSION 3.25)
project(tha_native_media LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit 12.8 REQUIRED)
set(THA_CUDA_INCLUDE_OVERLAY "" CACHE PATH "Previously qualified Fedora/CUDA header compatibility overlay")
set(THA_PTX_OPTIONS)
if(THA_CUDA_INCLUDE_OVERLAY)
  list(APPEND THA_PTX_OPTIONS -I${THA_CUDA_INCLUDE_OVERLAY})
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/media_kernels.ptx
  COMMAND ${CMAKE_CUDA_COMPILER} --ptx -arch=compute_89 -std=c++17
    -ccbin ${CMAKE_CUDA_HOST_COMPILER} ${THA_PTX_OPTIONS} ${CMAKE_CURRENT_SOURCE_DIR}/kernels.cu
    -o ${CMAKE_CURRENT_BINARY_DIR}/media_kernels.ptx
  DEPENDS kernels.cu VERBATIM)
add_custom_target(tha_media_kernels DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/media_kernels.ptx)
add_library(tha_gpu_encoder STATIC gpu_encoder.cpp)
target_include_directories(tha_gpu_encoder PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tha_gpu_encoder PUBLIC CUDA::cuda_driver PkgConfig::AV)
target_compile_options(tha_gpu_encoder PRIVATE -Wall -Wextra -Werror)
add_executable(tha_encoder_probe encoder_probe.cpp)
add_dependencies(tha_encoder_probe tha_media_kernels)
target_link_libraries(tha_encoder_probe PRIVATE tha_gpu_encoder)
add_executable(tha_encoder_contract encoder_contract.cpp)
add_dependencies(tha_encoder_contract tha_media_kernels)
target_link_libraries(tha_encoder_contract PRIVATE tha_gpu_encoder)
add_executable(tha_mesh_probe mesh_probe.cpp)
add_dependencies(tha_mesh_probe tha_media_kernels)
target_link_libraries(tha_mesh_probe PRIVATE tha_gpu_encoder)
set(THA_LIVEKIT_SOURCE "" CACHE PATH "Pinned LiveKit v1.11.0 source")
set(THA_LIVEKIT_BUILD "" CACHE PATH "Qualified C++/FFI library directory")
if(THA_LIVEKIT_SOURCE AND THA_LIVEKIT_BUILD)
  add_library(tha_livekit SHARED IMPORTED)
  set_target_properties(tha_livekit PROPERTIES IMPORTED_LOCATION ${THA_LIVEKIT_BUILD}/lib/liblivekit.so
    INTERFACE_INCLUDE_DIRECTORIES ${THA_LIVEKIT_SOURCE}/include)
  add_executable(tha_media_publisher publisher_probe.cpp)
  add_dependencies(tha_media_publisher tha_media_kernels)
  target_link_libraries(tha_media_publisher PRIVATE tha_gpu_encoder tha_livekit)
  set_target_properties(tha_media_publisher PROPERTIES BUILD_RPATH ${THA_LIVEKIT_BUILD}/lib)
endif()
