cmake_minimum_required(VERSION 3.25)
project(tha_native_media LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit 12.8 REQUIRED)
set(THA_CUDA_INCLUDE_OVERLAY "" CACHE PATH "Previously qualified Fedora/CUDA header compatibility overlay")
set(THA_PTX_OPTIONS)
if(THA_CUDA_INCLUDE_OVERLAY)
  list(APPEND THA_PTX_OPTIONS -I${THA_CUDA_INCLUDE_OVERLAY})
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/media_kernels.ptx
  COMMAND ${CMAKE_CUDA_COMPILER} --ptx -arch=compute_89 -std=c++17
    -ccbin ${CMAKE_CUDA_HOST_COMPILER} ${THA_PTX_OPTIONS} ${CMAKE_CURRENT_SOURCE_DIR}/kernels.cu
    -o ${CMAKE_CURRENT_BINARY_DIR}/media_kernels.ptx
  DEPENDS kernels.cu VERBATIM)
add_custom_target(tha_media_kernels DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/media_kernels.ptx)
add_library(tha_gpu_encoder STATIC gpu_encoder.cpp)
target_include_directories(tha_gpu_encoder PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tha_gpu_encoder PUBLIC CUDA::cuda_driver PkgConfig::AV)
target_compile_options(tha_gpu_encoder PRIVATE -Wall -Wextra -Werror)
add_executable(tha_encoder_probe encoder_probe.cpp)
add_dependencies(tha_encoder_probe tha_media_kernels)
target_link_libraries(tha_encoder_probe PRIVATE tha_gpu_encoder)
add_executable(tha_encoder_contract encoder_contract.cpp)
add_dependencies(tha_encoder_contract tha_media_kernels)
target_link_libraries(tha_encoder_contract PRIVATE tha_gpu_encoder)
add_executable(tha_mesh_probe mesh_probe.cpp)
add_dependencies(tha_mesh_probe tha_media_kernels)
target_link_libraries(tha_mesh_probe PRIVATE tha_gpu_encoder)
set(THA_LIVEKIT_SOURCE "" CACHE PATH "Pinned LiveKit v1.11.0 source")
set(THA_LIVEKIT_BUILD "" CACHE PATH "Qualified C++/FFI library directory")
if(THA_LIVEKIT_SOURCE AND THA_LIVEKIT_BUILD)
  add_library(tha_livekit SHARED IMPORTED)
  set_target_properties(tha_livekit PROPERTIES IMPORTED_LOCATION ${THA_LIVEKIT_BUILD}/lib/liblivekit.so
    INTERFACE_INCLUDE_DIRECTORIES ${THA_LIVEKIT_SOURCE}/include)
  add_executable(tha_media_publisher publisher_probe.cpp)
  add_dependencies(tha_media_publisher tha_media_kernels)
  target_link_libraries(tha_media_publisher PRIVATE tha_gpu_encoder tha_livekit)
  set_target_properties(tha_media_publisher PROPERTIES BUILD_RPATH ${THA_LIVEKIT_BUILD}/lib)
  add_executable(tha_session_media session_worker.cpp)
  add_dependencies(tha_session_media tha_media_kernels)
  target_link_libraries(tha_session_media PRIVATE tha_gpu_encoder tha_livekit)
  target_compile_options(tha_session_media PRIVATE -Wall -Wextra -Werror)
  set_target_properties(tha_session_media PROPERTIES BUILD_RPATH ${THA_LIVEKIT_BUILD}/lib)
endif()
add_executable(tha_large_mesh_probe large_probe.cpp)
add_dependencies(tha_large_mesh_probe tha_media_kernels)
target_link_libraries(tha_large_mesh_probe PRIVATE tha_gpu_encoder)
target_compile_options(tha_large_mesh_probe PRIVATE -Wall -Wextra -Werror)
set(THA_CLAIRE_SDK_BUILD "" CACHE PATH "Pinned Audio2Face SDK build for direct GPU research qualification")
set(THA_TENSORRT_ROOT "" CACHE PATH "Verified isolated TensorRT runtime")
if(THA_CLAIRE_SDK_BUILD AND THA_TENSORRT_ROOT)
  # Upstream SDK requires the old libstdc++ string ABI. Keep the existing
  # standalone/reference encoder ABI and build an explicit matching variant.
  add_library(tha_gpu_encoder_sdk STATIC gpu_encoder.cpp)
  target_include_directories(tha_gpu_encoder_sdk PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(tha_gpu_encoder_sdk PUBLIC _GLIBCXX_USE_CXX11_ABI=0)
  set_target_properties(tha_gpu_encoder_sdk PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(tha_gpu_encoder_sdk PUBLIC CUDA::cuda_driver PkgConfig::AV)
  target_compile_options(tha_gpu_encoder_sdk PRIVATE -Wall -Wextra -Werror)
  add_executable(tha_claire_render ../motion/render_probe.cpp)
  add_dependencies(tha_claire_render tha_media_kernels)
  target_include_directories(tha_claire_render SYSTEM PRIVATE ${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/include)
  target_link_libraries(tha_claire_render PRIVATE tha_gpu_encoder_sdk CUDA::cudart ${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/lib/libaudio2x.so)
  target_compile_options(tha_claire_render PRIVATE -Wall -Wextra -Werror)
  set_target_properties(tha_claire_render PROPERTIES BUILD_RPATH "${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/lib;${THA_TENSORRT_ROOT}/lib;/usr/local/cuda-12.8/lib64")
  add_library(tha_claire_bridge SHARED ../motion/claire_bridge.cpp)
  add_dependencies(tha_claire_bridge tha_media_kernels)
  target_include_directories(tha_claire_bridge SYSTEM PRIVATE ${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/include)
  target_link_libraries(tha_claire_bridge PRIVATE tha_gpu_encoder_sdk CUDA::cudart ${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/lib/libaudio2x.so)
  target_compile_options(tha_claire_bridge PRIVATE -Wall -Wextra -Werror)
  target_link_options(tha_claire_bridge PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/../motion/claire_bridge.map")
  set_property(TARGET tha_claire_bridge APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../motion/claire_bridge.map)
  set_target_properties(tha_claire_bridge PROPERTIES BUILD_RPATH "${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/lib;${THA_TENSORRT_ROOT}/lib;/usr/local/cuda-12.8/lib64")
  add_executable(tha_claire_bridge_probe ../motion/claire_bridge_probe.cpp)
  target_link_libraries(tha_claire_bridge_probe PRIVATE tha_claire_bridge)
  target_compile_options(tha_claire_bridge_probe PRIVATE -Wall -Wextra -Werror)
  if(THA_LIVEKIT_SOURCE AND THA_LIVEKIT_BUILD)
    add_executable(tha_claire_session_media claire_session_worker.cpp)
    target_link_libraries(tha_claire_session_media PRIVATE tha_claire_bridge tha_livekit ${CMAKE_DL_LIBS})
    target_compile_options(tha_claire_session_media PRIVATE -Wall -Wextra -Werror)
    set_target_properties(tha_claire_session_media PROPERTIES BUILD_RPATH "${THA_LIVEKIT_BUILD}/lib")
  endif()
endif()
add_executable(tha_large_mesh_contract large_contract.cpp)
add_executable(tha_publish_trace_contract publish_trace_contract.cpp)
add_executable(tha_probe_rate_contract probe_rate_contract.cpp)
target_compile_options(tha_probe_rate_contract PRIVATE -Wall -Wextra -Werror)
target_link_libraries(tha_publish_trace_contract PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(tha_publish_trace_contract PRIVATE -Wall -Wextra -Werror)
add_dependencies(tha_large_mesh_contract tha_media_kernels)
target_link_libraries(tha_large_mesh_contract PRIVATE tha_gpu_encoder)
target_compile_options(tha_large_mesh_contract PRIVATE -Wall -Wextra -Werror)
