cmake_minimum_required(VERSION 3.25)
project(tha_claire_motion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(CUDAToolkit 12.8 REQUIRED)
set(THA_TENSORRT_ROOT "" CACHE PATH "Verified isolated TensorRT 10.13.3.9")
if(NOT THA_TENSORRT_ROOT)
  message(FATAL_ERROR "Explicit verified TensorRT root required")
endif()
add_executable(tha_claire_engine engine_builder.cpp)
target_include_directories(tha_claire_engine SYSTEM PRIVATE ${THA_TENSORRT_ROOT}/include)
target_link_libraries(tha_claire_engine PRIVATE CUDA::cudart ${THA_TENSORRT_ROOT}/lib/libnvinfer.so.10 ${THA_TENSORRT_ROOT}/lib/libnvonnxparser.so.10)
target_compile_options(tha_claire_engine PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations)
set_target_properties(tha_claire_engine PROPERTIES BUILD_RPATH "${THA_TENSORRT_ROOT}/lib;/usr/local/cuda-12.8/lib64")
set(THA_CLAIRE_SDK_BUILD "" CACHE PATH "Qualified pinned Audio2Face SDK build")
if(THA_CLAIRE_SDK_BUILD)
  add_executable(tha_claire_geometry geometry_probe.cpp)
  target_include_directories(tha_claire_geometry SYSTEM PRIVATE ${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/include)
  target_compile_definitions(tha_claire_geometry PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
  target_link_libraries(tha_claire_geometry PRIVATE CUDA::cudart ${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/lib/libaudio2x.so)
  target_compile_options(tha_claire_geometry PRIVATE -Wall -Wextra -Werror)
  set_target_properties(tha_claire_geometry PROPERTIES BUILD_RPATH "${THA_CLAIRE_SDK_BUILD}/audio2x-sdk/lib;${THA_TENSORRT_ROOT}/lib;/usr/local/cuda-12.8/lib64")
endif()
